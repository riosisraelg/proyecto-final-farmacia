% 2. QUE EMPRESA REPRESENTAMOS
% =====================================================================
\section{La empresa}
\begin{frame}{¿Qué empresa representamos?}
  \textbf{Quiénes somos}
  \begin{itemize}
    \item Consultorio médico con farmacia asociada.
  \end{itemize}
  \vspace{0.3em}
  \textbf{Qué áreas intervienen}
  \begin{itemize}
    \item Consultorio (médicos), farmacia (dispensación), compras (proveedores) y administración.
  \end{itemize}
  \vspace{0.3em}
  \textbf{Cómo se relacionan el consultorio y la farmacia}
  \begin{itemize}
    \item El médico consulta y emite la receta; la farmacia la surte y registra la venta.
  \end{itemize}
\end{frame}

% =====================================================================
% 3. PROBLEMATICA ACTUAL
% =====================================================================
\section{Problemática actual}
\begin{frame}{¿Cuál es la problemática actual?}
  \textbf{Situación actual}
  \begin{itemize}
    \item Desconexión entre la farmacia y el consultorio.
    \item Operan como dos negocios distintos: información y procesos separados.
    \item Experiencia de usuario fragmentada: sin datos de fidelidad ni conocimiento de la conversión de recetas.
  \end{itemize}
  \textbf{Manejo de la información}
  \begin{itemize}
    \item La información se administra de forma arcaica.
    \item Uso de papel; datos dispersos, no centralizados ni conectados para generar información estratégica.
  \end{itemize}
  \textbf{Principales problemas identificados}
  \begin{itemize}
    \item Ausencia de automatización.
    \item Desabasto de medicamentos clave y mermas por caducidad no detectada.
    \item Procesos manuales que retrasan la reposición y provocan errores de dispensación.
  \end{itemize}
\end{frame}

% =====================================================================
% 4. IMPACTO
% =====================================================================
\section{Impacto}
\begin{frame}{¿Cuál es el impacto?}
  \textbf{Impacto en los pacientes}
  \begin{itemize}
    \item Experiencias distintas entre consultorio y farmacia.
    \item Creación de doble perfil.
  \end{itemize}
  \vspace{0.3em}
  \textbf{Impacto en la atención}
  \begin{itemize}
    \item Atención lenta y fragmentada.
  \end{itemize}
  \vspace{0.3em}
  \textbf{Impacto en la farmacia}
  \begin{itemize}
    \item Desconocimiento del rendimiento del negocio.
    \item Pérdida de ventas.
    \item Ausencia de competitividad.
  \end{itemize}
\end{frame}

% =====================================================================
% 5. CAUSA RAIZ
% =====================================================================
\section{Causa raíz}
\begin{frame}{¿Cuál es la causa raíz?}
  \textbf{Falta de integración de la información}
  \begin{itemize}
    \item Consultorio y farmacia no comparten una misma fuente de datos.
    \item Información dispersa y en papel: se duplica y no se conecta.
  \end{itemize}
  \vspace{0.3em}
  \textbf{Procesos manuales}
  \begin{itemize}
    \item Operación dependiente de tareas manuales, sin conexión entre áreas.
    \item No hay un flujo estandarizado de la información.
  \end{itemize}
  \vspace{0.3em}
  \textbf{Falta de validaciones y controles}
  \begin{itemize}
    \item Sin reglas de negocio aplicadas de forma automática.
    \item Sin control de inventario, caducidades ni trazabilidad receta--venta.
  \end{itemize}
\end{frame}

% =====================================================================
% 6. ESTRATEGIA A SEGUIR
% =====================================================================
\section{Estrategia}
\begin{frame}{¿Qué estrategia debemos seguir?}
  \begin{itemize}
    \item \textbf{Entender las reglas del negocio:} qué se valida, qué se controla.
    \item \textbf{Mapear el proceso actual:} de la consulta a la venta y la reposición.
    \item \textbf{Estandarizar la operación:} un flujo único y ordenado.
    \item \textbf{Centralizar la información:} una sola fuente de verdad.
    \item \textbf{Automatizar los procesos:} validaciones y registros sin intervención manual.
  \end{itemize}
  \vspace{0.3em}
  \small\textit{Primero la operación, luego la tecnología.}
\end{frame}

% =====================================================================
% 7. SOLUCION TECNOLOGICA - ER a PANTALLA COMPLETA
% =====================================================================
\section{Solución tecnológica}
\begin{frame}{¿Cuál es nuestra solución tecnológica?}
  \begin{itemize}
    \item \textbf{Base de datos relacional en MySQL:} centraliza toda la información.
    \item \textbf{Modelo Entidad-Relación:} 20 tablas con claves y relaciones definidas.
    \item \textbf{Integración de consultorio y farmacia:} un solo sistema, sin islas de datos.
  \end{itemize}
  \vspace{0.3em}
  \small\textit{Evidencia: Modelo Entidad-Relación (siguiente diapositiva).}
\end{frame}

% =====================================================================
% 8. COMO SE GESTIONAN LOS DATOS
% =====================================================================
\section{Gestión de los datos}
\begin{frame}{¿Cómo se gestionan los datos?}
  \textbf{Flujo de la información}
  \begin{itemize}
    \item Paciente $\rightarrow$ consulta $\rightarrow$ receta $\rightarrow$ surtido $\rightarrow$ venta $\rightarrow$ inventario $\rightarrow$ reposición.
  \end{itemize}
  \vspace{0.2em}
  \textbf{Información clínica}
  \begin{itemize}
    \item Historial, diagnósticos, alergias y contraindicaciones del paciente.
  \end{itemize}
  \textbf{Recetas y medicamentos}
  \begin{itemize}
    \item Recetas vinculadas a consulta, medicamento, dosis y duración.
  \end{itemize}
  \textbf{Inventario y ventas}
  \begin{itemize}
    \item Stock por lote, caducidades, ventas y su vínculo con la receta.
  \end{itemize}
\end{frame}

% =====================================================================
% 9. COMO AUTOMATIZAMOS LOS PROCESOS
% =====================================================================
\section{Automatización}
\begin{frame}{¿Cómo automatizamos los procesos?}
  \begin{itemize}
    \item \textbf{Vistas:} historial clínico, inventario con alertas y medicamentos más recetados/surtidos.
    \item \textbf{Stored Procedures:} registrar consulta/receta, surtir receta y generar reposición.
    \item \textbf{Triggers:} validan recetas y disparan la auditoría de inventario.
    \item \textbf{LOG de auditoría:} tabla \texttt{Auditoria\_Inventario} con cada movimiento de stock.
  \end{itemize}
  \evidencia{resultados reales de la BD (vistas / SP / triggers)}
\end{frame}

% =====================================================================
% 10. COMO APLICAMOS LAS REGLAS DE NEGOCIO
% =====================================================================
\section{Reglas de negocio}
\begin{frame}{¿Cómo aplicamos las reglas de negocio?}
  \begin{itemize}
    \item \textbf{Validación de pacientes y médicos:} paciente activo y médico con cédula vigente.
    \item \textbf{Alergias y contraindicaciones:} se verifican contra los medicamentos recetados.
    \item \textbf{Interacciones medicamentosas:} se bloquean combinaciones peligrosas.
    \item \textbf{Validación de recetas:} dosis, duración y vigencia dentro de rango.
  \end{itemize}
  \evidencia{mensaje real de validación/error}
\end{frame}

% =====================================================================
% 11. COMO CONTROLAMOS EL INVENTARIO
% =====================================================================
\section{Control de inventario}
\begin{frame}{¿Cómo controlamos el inventario?}
  \begin{itemize}
    \item \textbf{Stock de medicamentos:} estado suficiente / bajo / crítico / agotado.
    \item \textbf{Lotes y caducidades:} se dispensa el lote no vencido más próximo a caducar.
    \item \textbf{Reposición:} órdenes automáticas al llegar al stock mínimo.
    \item \textbf{Auditoría de movimientos:} cada cambio de stock queda registrado.
  \end{itemize}
  \evidencia{inventario + LOG de auditoría}
\end{frame}

% =====================================================================
% 12. COMO INTEGRAMOS OPERACION Y TECNOLOGIA
% =====================================================================
\section{Integración operación--tecnología}
\begin{frame}{¿Cómo integramos operación y tecnología?}
  \begin{itemize}
    \item \textbf{Proceso operativo:} el flujo real del negocio, estandarizado.
    \item \textbf{Reglas de negocio:} llevadas a restricciones, SP y triggers.
    \item \textbf{Base de datos:} MySQL como fuente única de verdad.
    \item \textbf{Conexión Java--MySQL:} acceso a los datos desde la aplicación (JDBC).
  \end{itemize}
  \evidencia{conexión y consulta desde Java (\texttt{App.java})}
\end{frame}

% =====================================================================
% 13. IMPORTANCIA DE UNA ESTRATEGIA INTEGRAL
% =====================================================================
\section{Estrategia integral}
\begin{frame}{¿Por qué es importante una estrategia integral?}
  \begin{itemize}
    \item \textbf{Entender las reglas de negocio:} define qué debe validar el sistema.
    \item \textbf{Mapear correctamente el flujo de información:} evita islas de datos.
    \item \textbf{Integrar operación y tecnología:} una sola fuente de verdad.
    \item \textbf{Gestionar la correcta ejecución:} vistas, SP y triggers hacen cumplir las reglas.
  \end{itemize}
  \vspace{0.4em}
  \small\textit{Responde directamente a los cuatro puntos de la exposición oral.}
\end{frame}

% 14. BENEFICIOS

\section{Beneficios}
\begin{frame}{¿Qué beneficios obtenemos?}
  \begin{itemize}
    \item \textbf{Seguridad del paciente:} validaciones clínicas automáticas.
    \item \textbf{Eficiencia operativa:} procesos estandarizados y más rápidos.
    \item \textbf{Control del inventario:} menos desabasto y menos mermas.
    \item \textbf{Automatización:} menos errores manuales.
    \item \textbf{Información para la toma de decisiones:} reportes de consumo y conversión.
  \end{itemize}
\end{frame}
